cmake_minimum_required(VERSION 3.20)
project(windows-firewall-explainer VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Windows only")
endif()
option(WFE_ANALYZE "Enable MSVC analysis" OFF)
option(WFE_CLANG_TIDY "Enable clang-tidy" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(wfe src/core.cpp src/policy.cpp)
target_include_directories(wfe PUBLIC src)
target_compile_definitions(wfe PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(wfe PUBLIC ole32 oleaut32 ws2_32)
add_executable(windows-firewall-explainer src/main.cpp src/version.rc)
target_link_libraries(windows-firewall-explainer PRIVATE wfe)
if(MSVC)
  target_compile_options(wfe PRIVATE /W4 /WX /utf-8)
  target_compile_options(windows-firewall-explainer PRIVATE /W4 /WX /utf-8)
  target_link_options(windows-firewall-explainer PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
  if(WFE_ANALYZE)
    target_compile_options(wfe PRIVATE /analyze)
    target_compile_options(windows-firewall-explainer PRIVATE /analyze)
  endif()
endif()
if(WFE_CLANG_TIDY)
  find_program(TIDY clang-tidy REQUIRED)
  set_target_properties(wfe windows-firewall-explainer PROPERTIES CXX_CLANG_TIDY "${TIDY};--warnings-as-errors=*")
endif()
enable_testing()
add_executable(wfe-tests tests/tests.cpp)
target_link_libraries(wfe-tests PRIVATE wfe)
if(MSVC)
  target_compile_options(wfe-tests PRIVATE /W4 /WX /utf-8)
endif()
add_test(NAME unit COMMAND wfe-tests)
add_test(NAME help COMMAND windows-firewall-explainer --help)
add_test(NAME version COMMAND windows-firewall-explainer --version)
add_test(NAME malformed COMMAND windows-firewall-explainer explain --protocol garbage)
set_tests_properties(malformed PROPERTIES WILL_FAIL TRUE)
find_program(PWSH pwsh)
if(PWSH)
  add_test(NAME integration COMMAND "${PWSH}" -NoProfile -File "${CMAKE_SOURCE_DIR}/tests/integration.ps1" -Executable $<TARGET_FILE:windows-firewall-explainer>)
  add_test(NAME safety COMMAND "${PWSH}" -NoProfile -File "${CMAKE_SOURCE_DIR}/tests/safety.ps1" -Root "${CMAKE_SOURCE_DIR}")
endif()
install(TARGETS windows-firewall-explainer RUNTIME DESTINATION .)
install(FILES README.md LICENSE VALIDATION.md DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(WFE_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
  set(WFE_ARCH "x64")
endif()
set(CPACK_PACKAGE_FILE_NAME "windows-firewall-explainer-1.0.0-windows-${WFE_ARCH}")
include(CPack)
